# Lido ETH Permit Message API 实现指南

## 概述

本文档提供 `/earn/v1/lido-eth/tx/permit_message` 接口的完整实现指南，用于生成 Lido ETH 提取所需的 EIP-712 签名消息。

---

## 1. 接口规范

### 1.1 接口信息

- **路径**: `POST /earn/v1/lido-eth/tx/permit_message`
- **Content-Type**: `application/json`
- **认证**: 需要用户认证（根据你的认证机制）

### 1.2 请求参数

```typescript
interface RequestBody {
  amount: string; // 提取金额（wei 格式，字符串）
  accountAddress: string; // 用户账户地址（0x 开头）
  networkId: string; // 网络 ID（如 "evm--1" 表示 Ethereum Mainnet）
}
```

**参数说明**：

- `amount`: 用户要提取的 ETH 数量，以 wei 为单位（1 ETH = 10^18 wei），字符串格式
- `accountAddress`: 执行提取的账户地址，必须是有效的以太坊地址
- `networkId`: 网络标识符，用于确定链 ID

### 1.3 响应结构

```typescript
interface Response {
  data: {
    message: string; // EIP-712 格式的签名消息（JSON 字符串）
    deadline: number; // 签名截止时间（Unix 时间戳，秒）
  };
}
```

**响应说明**：

- `message`: EIP-712 Typed Data V4 格式的 JSON 字符串，用于前端签名
- `deadline`: 签名的有效期截止时间，Unix 时间戳（秒），建议设置为当前时间 + 1 小时

---

## 2. EIP-712 消息结构

### 2.1 Lido ETH Unstake Permit 消息格式

根据 Lido 协议的要求，EIP-712 消息应包含以下结构：

```typescript
interface EIP712Message {
  types: {
    EIP712Domain: Array<{
      name: string;
      type: string;
    }>;
    Permit: Array<{
      name: string;
      type: string;
    }>;
  };
  domain: {
    name: string; // 通常是 "Lido" 或相关合约名称
    version: string; // 版本号，如 "1"
    chainId: number; // 链 ID（1 = Ethereum Mainnet）
    verifyingContract: string; // Lido 合约地址
  };
  primaryType: 'Permit';
  message: {
    owner: string; // 用户地址（accountAddress）
    spender: string; // 授权给谁（通常是 Lido 合约地址）
    value: string; // 提取金额（amount，字符串格式）
    nonce: string; // 用户 nonce（从链上获取）
    deadline: string; // 截止时间（deadline，字符串格式）
  };
}
```

### 2.2 关键字段说明

1. **domain.chainId**:

   - Ethereum Mainnet: `1`
   - 从 `networkId` 解析：`evm--1` → `1`

2. **domain.verifyingContract**:

   - Lido ETH 主网合约地址（需要查询 Lido 官方文档获取最新地址）
   - 示例：`0x...`（请使用 Lido 官方提供的地址）

3. **message.nonce**:

   - 需要从链上查询用户的当前 nonce
   - 调用 Lido 合约的 `nonces(address)` 方法获取

4. **message.deadline**:
   - 建议设置为：`Math.floor(Date.now() / 1000) + 3600`（当前时间 + 1 小时）
   - 确保有足够时间完成签名和交易

## 3. 实现步骤

### 3.1 参数验证

```typescript
function validateRequest(body: RequestBody): void {
  // 验证 amount
  if (!body.amount || !/^\d+$/.test(body.amount)) {
    throw new Error('Invalid amount: must be a positive integer string');
  }

  // 验证 accountAddress
  if (
    !body.accountAddress ||
    !/^0x[a-fA-F0-9]{40}$/.test(body.accountAddress)
  ) {
    throw new Error('Invalid accountAddress: must be a valid Ethereum address');
  }

  // 验证 networkId
  if (!body.networkId || !body.networkId.startsWith('evm--')) {
    throw new Error('Invalid networkId: must be an EVM network');
  }

  // 验证金额不为 0
  if (BigInt(body.amount) === 0n) {
    throw new Error('Amount must be greater than 0');
  }
}
```

### 3.2 解析网络信息

```typescript
function parseNetworkId(networkId: string): { chainId: number } {
  // 从 networkId 解析链 ID
  // evm--1 -> 1 (Ethereum Mainnet)
  // evm--5 -> 5 (Goerli Testnet)
  const match = networkId.match(/^evm--(\d+)$/);
  if (!match) {
    throw new Error(`Unsupported network: ${networkId}`);
  }

  const chainId = parseInt(match[1], 10);
  return { chainId };
}
```

### 3.3 获取 Lido 合约信息

```typescript
// Lido 合约地址（需要根据实际网络配置）
const LIDO_CONTRACTS = {
  1: {
    // Ethereum Mainnet
    address: '0x...', // Lido 合约地址（请查询 Lido 官方文档）
    name: 'Lido',
    version: '1',
  },
  // 可以添加其他网络
};

function getLidoContractInfo(chainId: number) {
  const contract = LIDO_CONTRACTS[chainId];
  if (!contract) {
    throw new Error(`Lido not supported on chain ${chainId}`);
  }
  return contract;
}
```

### 3.4 查询用户 Nonce

```typescript
async function getUserNonce(
  accountAddress: string,
  lidoContractAddress: string,
  chainId: number,
): Promise<string> {
  // 使用 Web3 或 ethers.js 调用合约
  // 调用 Lido 合约的 nonces(address) 方法

  // 示例（使用 ethers.js）:
  const provider = getProvider(chainId);
  const contract = new ethers.Contract(
    lidoContractAddress,
    ['function nonces(address owner) view returns (uint256)'],
    provider,
  );

  const nonce = await contract.nonces(accountAddress);
  return nonce.toString();
}
```

### 3.5 计算 Deadline

```typescript
function calculateDeadline(): number {
  // 当前时间 + 1 小时（3600 秒）
  return Math.floor(Date.now() / 1000) + 3600;

  // 或者可以根据业务需求调整有效期
  // 例如：30 分钟 = 1800 秒
  // 例如：24 小时 = 86400 秒
}
```

### 3.6 构建 EIP-712 消息

```typescript
function buildEIP712Message(params: {
  accountAddress: string;
  amount: string;
  nonce: string;
  deadline: number;
  chainId: number;
  lidoContract: { address: string; name: string; version: string };
}): string {
  const { accountAddress, amount, nonce, deadline, chainId, lidoContract } =
    params;

  const message = {
    types: {
      EIP712Domain: [
        { name: 'name', type: 'string' },
        { name: 'version', type: 'string' },
        { name: 'chainId', type: 'uint256' },
        { name: 'verifyingContract', type: 'address' },
      ],
      Permit: [
        { name: 'owner', type: 'address' },
        { name: 'spender', type: 'address' },
        { name: 'value', type: 'uint256' },
        { name: 'nonce', type: 'uint256' },
        { name: 'deadline', type: 'uint256' },
      ],
    },
    domain: {
      name: lidoContract.name,
      version: lidoContract.version,
      chainId: chainId,
      verifyingContract: lidoContract.address,
    },
    primaryType: 'Permit',
    message: {
      owner: accountAddress,
      spender: lidoContract.address, // 通常是 Lido 合约地址
      value: amount,
      nonce: nonce,
      deadline: deadline.toString(),
    },
  };

  // 返回 JSON 字符串
  return JSON.stringify(message);
}
```

### 3.7 完整实现示例

```typescript
import { ethers } from 'ethers';
import { getProvider } from './provider'; // 你的 provider 获取函数

// Lido 合约配置
const LIDO_CONTRACTS = {
  1: {
    address: '0x...', // 请查询 Lido 官方文档获取最新地址
    name: 'Lido',
    version: '1',
  },
};

async function handleLidoEthPermitMessage(req: Request, res: Response) {
  try {
    // 1. 参数验证
    const { amount, accountAddress, networkId } = req.body;
    validateRequest({ amount, accountAddress, networkId });

    // 2. 解析网络信息
    const { chainId } = parseNetworkId(networkId);

    // 3. 获取 Lido 合约信息
    const lidoContract = getLidoContractInfo(chainId);

    // 4. 查询用户 nonce
    const nonce = await getUserNonce(
      accountAddress,
      lidoContract.address,
      chainId,
    );

    // 5. 计算 deadline
    const deadline = calculateDeadline();

    // 6. 构建 EIP-712 消息
    const message = buildEIP712Message({
      accountAddress,
      amount,
      nonce,
      deadline,
      chainId,
      lidoContract,
    });

    // 7. 返回结果
    res.json({
      data: {
        message,
        deadline,
      },
    });
  } catch (error) {
    res.status(400).json({
      error: error.message,
    });
  }
}
```

## 4. 安全考虑

### 4.1 参数验证

- ✅ 验证 `amount` 必须是正整数字符串
- ✅ 验证 `accountAddress` 必须是有效的以太坊地址格式
- ✅ 验证 `networkId` 必须是支持的 EVM 网络
- ✅ 验证金额不能为 0

### 4.2 防重放攻击

- ✅ 使用 `nonce` 确保每次签名的唯一性
- ✅ 设置合理的 `deadline`，防止签名被长期使用
- ✅ 在 `/earn/v2/unstake` 接口中验证签名和 deadline

### 4.3 金额验证

- ✅ 验证用户是否有足够的余额（可选，在 `/earn/v2/unstake` 中验证更合适）
- ✅ 验证金额格式正确（wei 格式）

### 4.4 网络验证

- ✅ 只支持 Lido 已部署的网络
- ✅ 验证链 ID 与 networkId 匹配

## 6. 错误处理

```typescript
enum ErrorCode {
  INVALID_AMOUNT = 'INVALID_AMOUNT',
  INVALID_ADDRESS = 'INVALID_ADDRESS',
  UNSUPPORTED_NETWORK = 'UNSUPPORTED_NETWORK',
  CONTRACT_ERROR = 'CONTRACT_ERROR',
  INSUFFICIENT_BALANCE = 'INSUFFICIENT_BALANCE',
}

function handleError(error: Error): Response {
  if (error.message.includes('amount')) {
    return { code: ErrorCode.INVALID_AMOUNT, message: error.message };
  }
  if (error.message.includes('address')) {
    return { code: ErrorCode.INVALID_ADDRESS, message: error.message };
  }
  if (error.message.includes('network')) {
    return { code: ErrorCode.UNSUPPORTED_NETWORK, message: error.message };
  }
  return { code: ErrorCode.CONTRACT_ERROR, message: error.message };
}
```

## 10. 完整代码示例（Node.js + Express）

```typescript
import express from 'express';
import { ethers } from 'ethers';
import {
  validateRequest,
  parseNetworkId,
  getLidoContractInfo,
  calculateDeadline,
  buildEIP712Message,
} from './utils';

const app = express();
app.use(express.json());

// Provider 配置
const providers = {
  1: new ethers.JsonRpcProvider('https://eth.llamarpc.com'), // Ethereum Mainnet
  // 添加其他网络
};

// Lido 合约配置
const LIDO_CONTRACTS = {
  1: {
    address: '0x...', // 请查询 Lido 官方文档
    name: 'Lido',
    version: '1',
  },
};

// 获取用户 nonce
async function getUserNonce(
  accountAddress: string,
  contractAddress: string,
  chainId: number,
): Promise<string> {
  const provider = providers[chainId];
  if (!provider) {
    throw new Error(`Provider not configured for chain ${chainId}`);
  }

  const contract = new ethers.Contract(
    contractAddress,
    ['function nonces(address owner) view returns (uint256)'],
    provider,
  );

  const nonce = await contract.nonces(accountAddress);
  return nonce.toString();
}

// 接口实现
app.post('/earn/v1/lido-eth/tx/permit_message', async (req, res) => {
  try {
    const { amount, accountAddress, networkId } = req.body;

    // 1. 参数验证
    validateRequest({ amount, accountAddress, networkId });

    // 2. 解析网络
    const { chainId } = parseNetworkId(networkId);

    // 3. 获取合约信息
    const lidoContract = getLidoContractInfo(chainId);

    // 4. 查询 nonce
    const nonce = await getUserNonce(
      accountAddress,
      lidoContract.address,
      chainId,
    );

    // 5. 计算 deadline
    const deadline = calculateDeadline();

    // 6. 构建消息
    const message = buildEIP712Message({
      accountAddress,
      amount,
      nonce,
      deadline,
      chainId,
      lidoContract,
    });

    // 7. 返回结果
    res.json({
      data: {
        message,
        deadline,
      },
    });
  } catch (error) {
    console.error('Error generating permit message:', error);
    res.status(400).json({
      error: error.message,
    });
  }
});

app.listen(3000, () => {
  console.log('Server running on port 3000');
});
```
